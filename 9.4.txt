#include <cstdlib>
#include <iostream>
using namespace std;
int main() {
    char car;
    bool digito_leido = false; // no se ha leído ningún dato
    while (!digito_leido) {
        cout << "Introduzca un caracter digito para salir del bucle: ";
        cin >> car;
        digito_leido = (('0' <= car) && (car <= '9'));
    } // fin de while
    cout << car << " es el digito leido" << endl;
    system("PAUSE");
    return EXIT_SUCCESS;
}
